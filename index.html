import streamlit as st import pandas as pd import numpy as np import matplotlib.pyplot as plt import seaborn as sns from supabase import create_client, Client import os # --- CONFIGURAÇÃO DA PÁGINA --- st.set_page_config( page_title="BioData Analyzer - UFRJ", page_icon="🩸", layout="wide" ) # --- CONEXÃO COM O BANCO DE DADOS (SUPABASE) --- @st.cache_resource def iniciar_conexao(): url = os.environ.get("SUPABASE_URL") key = os.environ.get("SUPABASE_KEY") if not url or not key: try: url = url or st.secrets.get("SUPABASE_URL") key = key or st.secrets.get("SUPABASE_KEY") except: pass if not url or not key: return None try: return create_client(url, key) except: return None supabase = iniciar_conexao() # --- FUNÇÃO DE BUSCA DE DADOS --- def carregar_dados(): if not supabase: return pd.DataFrame() try: resposta = supabase.table("registros_treino").select("*").execute() return pd.DataFrame(resposta.data) if resposta.data else pd.DataFrame() except Exception as e: st.error(f"⚠️ Erro de conexão com o banco de dados: {e}") return pd.DataFrame() # --- INICIALIZAÇÃO DE ESTADOS --- if 'logado' not in st.session_state: st.session_state.logado = False st.session_state.perfil = None st.session_state.nome_usuario = None st.session_state.id_usuario = None if 'admin_logado' not in st.session_state: st.session_state.admin_logado = False # ========================================== # NAVEGAÇÃO PRINCIPAL (MENU LATERAL) # ========================================== st.sidebar.title("Navegação") menu = st.sidebar.radio("Selecione a Área:", ["📊 Painel Principal", "⚙️ Administração"]) st.sidebar.divider() # ========================================== # ÁREA 1: ADMINISTRAÇÃO # ========================================== if menu == "⚙️ Administração": st.title("⚙️ Administração do Sistema") if not st.session_state.admin_logado: with st.form(key='admin_login'): senha_admin_input = st.text_input("Senha de Administrador:", type="password") if st.form_submit_button("Acessar"): if senha_admin_input == os.environ.get("SENHA_ADMIN", "").strip(): st.session_state.admin_logado = True st.rerun() else: st.error("❌ Senha incorreta.") else: st.success("🟢 Modo Administrador Ativo") if st.button("🔒 Sair"): st.session_state.admin_logado = False st.rerun() st.subheader("Cadastrar Novo Usuário") with st.form(key="form_cad", clear_on_submit=True): novo_id = st.text_input("ID do Usuário:") novo_nome = st.text_input("Nome Completo:") novo_perfil = st.selectbox("Perfil:", ["Pesquisador (Inserção)", "Professor/PhD (Análise)", "Comitê de Ética (Auditoria)"]) nova_senha = st.text_input("Senha:", type="password") if st.form_submit_button("Salvar Usuário"): try: supabase.table("usuarios").insert({"id_usuario": novo_id, "nome": novo_nome, "perfil": novo_perfil, "senha": nova_senha}).execute() st.success("✅ Usuário cadastrado!") except Exception as e: st.error(f"Erro: {e}") # ========================================== # ÁREA 2: PAINEL PRINCIPAL # ========================================== elif menu == "📊 Painel Principal": if not st.session_state.logado: with st.sidebar.form(key='login_form'): perfil_sel = st.selectbox("Selecione seu Perfil:", ["Selecionar...", "Pesquisador (Inserção)", "Professor/PhD (Análise)", "Comitê de Ética (Auditoria)"]) id_in = st.text_input("ID do Usuário:") sen_in = st.text_input("Senha:", type="password") if st.form_submit_button("Entrar"): res = supabase.table("usuarios").select("*").eq("id_usuario", id_in).eq("perfil", perfil_sel).eq("senha", sen_in).execute() if res.data: u = res.data[0] st.session_state.update({'logado': True, 'perfil': u['perfil'], 'nome_usuario': u['nome'], 'id_usuario': u['id_usuario']}) st.rerun() else: st.error("❌ Dados inválidos.") else: if st.sidebar.button("Logout"): for k in ['logado', 'perfil', 'nome_usuario', 'id_usuario']: st.session_state[k] = None st.rerun() perfil = st.session_state.perfil if perfil in ["Professor/PhD (Análise)", "Comitê de Ética (Auditoria)"]: st.title(f"📊 Painel de Dados - {perfil}") filtro_genero = st.selectbox("Visualização:", ["Geral (Ambos os Gêneros)", "Masculino", "Feminino"]) df_banco = carregar_dados() if not df_banco.empty: df_f = df_banco[df_banco['glicemia'] > 60] if filtro_genero != "Geral (Ambos os Gêneros)": df_f = df_f[df_f['genero'] == filtro_genero] m1, m2, m3 = st.columns(3) m1.metric("Média (Glicemia)", f"{df_f['glicemia'].mean():.1f} mg/dL") m3.metric("Total (N)", len(df_f)) st.dataframe(df_f, use_container_width=True) # Botão de Backup Físico st.download_button("Baixar Backup (CSV)", df_f.to_csv(index=False).encode('utf-8'), "backup_avaliacoes.csv", "text/csv") elif perfil == "Pesquisador (Inserção)": st.title("📥 Registro de Dados em Campo") with st.form(key='form_coleta', clear_on_submit=True): c1, c2 = st.columns(2) id_p = c1.text_input("ID do Sujeito:") dur = c1.number_input("Duração (min):", value=45) gen = c2.radio("Gênero:", ["Masculino", "Feminino"], horizontal=True) gli = c2.number_input("Glicemia (mg/dL):", value=100) int_val = c2.selectbox("Intensidade:", [1, 2, 3]) if st.form_submit_button("Salvar Registro Seguro"): dados = {"id_paciente": id_p, "genero": gen, "glicemia": gli, "duracao": dur, "intensidade": int_val} try: # DUAL INSERT: Tabela Principal + Tabela Backup supabase.table("registros_treino").insert(dados).execute() supabase.table("backup_registros_treino").insert(dados).execute() st.success("✅ Dados salvos com backup de segurança!") except Exception as e: st.error(f"❌ Erro ao salvar: {e}")
